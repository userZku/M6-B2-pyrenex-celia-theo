# Trigger réentraînement — vérifie toutes les 6h s'il y a au moins 200
# feedbacks non consommés (used_for_training = 0) ; sinon retrain.py sort en
# succès (skip, pas une erreur). Chemins absolus + venv : cron part de $HOME.
0 */6 * * * cd /opt/pyrenex && .venv/bin/python scripts/retrain.py --min-feedback 200 >> logs/retrain.log 2>&1

# Déclenchement manuel (hors attente des 6h) : `workflow_dispatch` sur le
